# i18n via next-intl Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add Russian and Ukrainian language support to the discovery site without breaking existing English URLs (`/`, `/survey`, `/privacy`, etc.) already published in promotional posts.

**Architecture:** next-intl v4 with `localePrefix: 'as-needed'` so EN stays at root. Routes move under `src/app/[locale]/`. Dictionaries live in `dictionary/{en,ru,uk}.json`. Legal pages (Privacy/Terms/Cookies/Disclaimer) always render English content via explicit `getTranslations({ locale: 'en' })`. A `LegalBanner` shows on `/ru/*` and `/uk/*` legal pages. Server-side `language` is stamped onto every survey response from `x-next-intl-locale` header set by `proxy.ts`.

**Tech Stack:** Next.js 16 (App Router, `proxy.ts`), React 19, next-intl ≥4.0, TypeScript, Tailwind 4, Supabase Postgres.

**Testing approach:** This project has no test suite (per CLAUDE.md). Each task verifies via `npm run typecheck`, `npm run build`, and manual browser smoke. Replace standard TDD red-green-refactor with **apply → typecheck → manual smoke → commit**.

**Commit convention:** Short imperative subject (`Add ...`, `Update ...`, `Move ...`, `Replace ...`). NO `Co-Authored-By` trailer — matches existing repo style.

### Task 4: Create request.ts (loads dictionaries)

**Files:**

- [ ] **Step 1: Write request.ts**

```ts
// src/i18n/request.ts
import { getRequestConfig } from "next-intl/server";
import { hasLocale } from "next-intl";
import { routing } from "./routing";

export default getRequestConfig(async ({ requestLocale }) => {
  const requested = await requestLocale;
  const locale = hasLocale(routing.locales, requested)
    ? requested
    : routing.defaultLocale;

  return {
    locale,
    messages: (await import(`../../dictionary/${locale}.json`)).default,
  };
});
```

- [ ] **Step 2: Note — this file will fail to resolve dictionary until Phase 2**

The `import()` will fail at runtime if `dictionary/{locale}.json` doesn't exist, but TypeScript won't catch it (dynamic import). Build will fail at Phase 3 when proxy.ts is added and routing kicks in.

- [ ] **Step 1: Create [locale]/layout.tsx**

This replaces the current `src/app/layout.tsx`. It now accepts `params: Promise<{ locale }>` and calls `setRequestLocale` for static rendering.

```tsx
// src/app/[locale]/layout.tsx
import type { Metadata, Viewport } from "next";
import type { ReactNode } from "react";
import { Fraunces, JetBrains_Mono, Caveat } from "next/font/google";
import { notFound } from "next/navigation";
import { hasLocale, NextIntlClientProvider } from "next-intl";
import { getTranslations, setRequestLocale } from "next-intl/server";
import { env } from "@/shared/config/env";
import { SurveyDraftProvider } from "@/features/multi-step-form";
import { routing } from "@/i18n/routing";
import "../globals.css";

const fraunces = Fraunces({
  subsets: ["latin"],
  weight: ["400", "500", "600", "700"],
  variable: "--font-fraunces",
  display: "swap",
});

const jetbrainsMono = JetBrains_Mono({
  subsets: ["latin"],
  weight: ["400", "500"],
  variable: "--font-jetbrains-mono",
  display: "swap",
});

const caveat = Caveat({
  subsets: ["latin"],
  weight: ["500", "600"],
  variable: "--font-caveat",
  display: "swap",
});

export async function generateMetadata({
  params,
}: {
  params: Promise<{ locale: string }>;
}): Promise<Metadata> {
  const { locale } = await params;
  const t = await getTranslations({ locale, namespace: "GlobalMeta" });

  return {
    metadataBase: new URL(env.NEXT_PUBLIC_SITE_URL ?? "http://localhost:3000"),
    title: t("title"),
    description: t("description"),
    openGraph: {
      title: t("title"),
      description: t("description"),
      type: "website",
    },
    twitter: {
      card: "summary_large_image",
      title: t("title"),
      description: t("description"),
    },
    robots: { index: true, follow: true },
  };
}

export const viewport: Viewport = {
  themeColor: "#f4f0e8",
};

export function generateStaticParams() {
  return routing.locales.map((locale) => ({ locale }));
}

type TRootLayoutProps = {
  children: ReactNode;
  params: Promise<{ locale: string }>;
};

export default async function RootLayout({
  children,
  params,
}: TRootLayoutProps) {
  const { locale } = await params;

  if (!hasLocale(routing.locales, locale)) {
    notFound();
  }

  setRequestLocale(locale);

  return (
    <html
      lang={locale}
      className={`${fraunces.variable} ${jetbrainsMono.variable} ${caveat.variable}`}
    >
      <body>
        <a
          href="#main"
          className="sr-only focus:not-sr-only focus:fixed focus:top-3 focus:left-3 focus:z-50 focus:bg-ink focus:text-paper focus:px-4 focus:py-2 focus:font-mono focus:text-xs focus:uppercase focus:tracking-[0.18em]"
        >
          Skip to content
        </a>
        <NextIntlClientProvider>
          <SurveyDraftProvider>
            <div className="container max-w-[720px] mx-auto px-6 relative z-10">
              {children}
            </div>
          </SurveyDraftProvider>
        </NextIntlClientProvider>
      </body>
    </html>
  );
}
```

```tsx
// src/shared/ui/locale-switcher/LocaleSwitcher.tsx
"use client";

import { useState, useRef, useEffect } from "react";
import { useLocale, useTranslations } from "next-intl";
import { Link, usePathname } from "@/i18n/navigation";
import { routing } from "@/i18n/routing";

export function LocaleSwitcher() {
  const active = useLocale();
  const pathname = usePathname();
  const t = useTranslations("LocaleSwitcher");
  const [open, setOpen] = useState(false);
  const ref = useRef<HTMLDivElement>(null);

  useEffect(() => {
    const onClick = (e: MouseEvent) => {
      if (!ref.current?.contains(e.target as Node)) setOpen(false);
    };
    const onEsc = (e: KeyboardEvent) => {
      if (e.key === "Escape") setOpen(false);
    };
    document.addEventListener("click", onClick);
    document.addEventListener("keydown", onEsc);
    return () => {
      document.removeEventListener("click", onClick);
      document.removeEventListener("keydown", onEsc);
    };
  }, []);

  return (
    <div ref={ref} className="relative">
      <button
        type="button"
        onClick={() => setOpen(!open)}
        className="font-mono text-[11px] uppercase tracking-[0.18em] text-ink-soft hover:text-rust transition-colors"
        aria-label={t("label")}
        aria-expanded={open}
        aria-haspopup="listbox"
      >
        {active.toUpperCase()} ▾
      </button>
      {open && (
        <ul
          role="listbox"
          className="absolute right-0 top-full mt-2 bg-paper border border-rule py-1 min-w-[140px] z-10 shadow-lg"
        >
          {routing.locales
            .filter((l) => l !== active)
            .map((l) => (
              <li key={l} role="option">
                <Link
                  href={pathname}
                  locale={l}
                  className="block px-3 py-2 font-mono text-[11px] uppercase tracking-[0.18em] text-ink-soft hover:text-rust hover:bg-bg transition-colors"
                  onClick={() => setOpen(false)}
                >
                  {t(l as "en" | "ru" | "uk")}
                </Link>
              </li>
            ))}
        </ul>
      )}
    </div>
  );
}
```

- [ ] **Step 2: Replace with locale-aware version**

```ts
// src/app/sitemap.ts
import type { MetadataRoute } from "next";
import { routing } from "@/i18n/routing";

const SITE = process.env.NEXT_PUBLIC_SITE_URL ?? "https://[PLACEHOLDER: site URL]";

export default function sitemap(): MetadataRoute.Sitemap {
  const localised = ["", "/survey"];
  const legal = ["/privacy", "/terms", "/cookies", "/disclaimer"];

  const entries: MetadataRoute.Sitemap = [];

  for (const path of localised) {
    for (const locale of routing.locales) {
      const prefix = locale === routing.defaultLocale ? "" : `/${locale}`;
      entries.push({
        url: `${SITE}${prefix}${path}`,
        alternates: {
          languages: Object.fromEntries(
            routing.locales.map((l) => [
              l,
              `${SITE}${l === routing.defaultLocale ? "" : `/${l}`}${path}`,
            ])
          ),
        },
      });
    }
  }

  for (const path of legal) {
    entries.push({ url: `${SITE}${path}` });
  }

  return entries;
}
```
